from collections import Counter

def top_word(text: str) -> str:
    words = Counter(text.lower().split())
    return words.most_common(1)[0][0]

story = "to be or not to be that is the question"
print(f"most common word: {top_word(story)!r}")

scores = {"Ada": 97, "Linus": 88, "Guido": 99}
honours = {n: "distinction" for n, s in scores.items() if s >= 90}
print(honours)
